-- Query to analyze user behavior
SELECT
  DATE_TRUNC('day', event_time) AS day,
  event_type,
  COUNT(*) AS total_events
FROM events
WHERE event_time >= CURRENT_DATE - INTERVAL '30 days'
GROUP BY 1, 2
ORDER BY 1, 2;
